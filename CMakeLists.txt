cmake_minimum_required(VERSION 3.20)
project(pushable_memory VERSION 0.1.0 LANGUAGES CXX)
add_library(pushable_memory INTERFACE)
target_include_directories(pushable_memory INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(pushable_memory INTERFACE cxx_std_20)
option(PUSHABLE_SANITIZE "Enable ASan and UBSan on GCC/Clang" OFF)
if(PUSHABLE_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pushable_memory INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pushable_memory INTERFACE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(reference_tests tests/reference_tests.cpp)
  target_link_libraries(reference_tests PRIVATE pushable_memory)
  add_test(NAME reference_tests COMMAND reference_tests)
  # Allocation replacement and sanitizer allocators should not be mixed.
  if(NOT PUSHABLE_SANITIZE)
    add_executable(no_heap tests/no_heap.cpp)
    target_link_libraries(no_heap PRIVATE pushable_memory)
    add_test(NAME no_heap COMMAND no_heap)
  endif()
endif()
add_executable(flex_document_demo examples/flex_document_demo.cpp)
target_link_libraries(flex_document_demo PRIVATE pushable_memory)
if(BUILD_TESTING)
  add_test(NAME flex_document_demo COMMAND flex_document_demo)
endif()
